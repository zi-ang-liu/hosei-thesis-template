% =====================================================================
%  サンプル図：数値実験の結果（pgfplots）
%  make figures で ja/figures/ と en/figures/ に PDF を出力する．
%  単独で試すとき：lualatex results.tex（日本語・ギャップ）
%  \figlang = ja / en，\figkind = gap（最適値とのギャップ）/ time（計算時間）
%  値は説明用に式で作った架空のデータ（本文の表 5.2 の n = 1000 の値と一致させている）．
%    ギャップ：貪欲法 1 + 2 exp(-n/300)，提案手法 0.2 + 0.6 exp(-n/300)
%    計算時間：厳密解法 0.83 (n/100)^2，貪欲法 0.001 (n/100)，提案手法 0.2 (n/100)
% =====================================================================
\documentclass[border=2pt]{standalone}
\usepackage{luatexja}
\usepackage[haranoaji]{luatexja-preset}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\providecommand*\figlang{ja}
\providecommand*\figkind{gap}
\def\langJA{ja}
\def\kindGap{gap}
\newcommand\JE[2]{\ifx\figlang\langJA #1\else #2\fi}

\definecolor{navy}{RGB}{0,82,155}
\definecolor{orange}{RGB}{214,104,26}
\definecolor{green}{RGB}{0,140,90}
\pgfplotsset{
  every axis/.append style={
    scale only axis, width=5.8cm, height=4.2cm,
    xlabel={\JE{品物数 $n$}{Number of items $n$}},
    xmin=50, xmax=1050, xtick={200,400,...,1000},
    grid=major, grid style={gray!20},
    tick label style={font=\small}, label style={font=\small},
    legend style={font=\small, draw=gray!50}, legend cell align=left,
    axis line style={gray!70},
    /pgf/number format/1000 sep={},
  },
  data/.style={thick, mark size=1.8pt, samples at={100,200,...,1000}},
}

\begin{document}
\begin{tikzpicture}
\ifx\figkind\kindGap
  \begin{axis}[ylabel={\JE{ギャップ [\%]}{Gap [\%]}}, ymin=0, ymax=3, legend pos=north east]
    \addplot[data, orange, dashed, mark=square*, mark options={solid}] {1 + 2*exp(-x/300)};
    \addlegendentry{\JE{貪欲法}{Greedy}}
    \addplot[data, navy, mark=*] {0.2 + 0.6*exp(-x/300)};
    \addlegendentry{\JE{提案手法}{Proposed}}
  \end{axis}
\else
  \begin{axis}[ylabel={\JE{計算時間 [s]}{Computation time [s]}},
               ymode=log, ymin=5e-4, ymax=1e4, legend pos=north west]
    \addplot[data, green, dashdotted, mark=triangle*, mark options={solid}] {0.83*(x/100)^2};
    \addlegendentry{\JE{厳密解法}{Exact (DP)}}
    \addplot[data, orange, dashed, mark=square*, mark options={solid}] {0.001*(x/100)};
    \addlegendentry{\JE{貪欲法}{Greedy}}
    \addplot[data, navy, mark=*] {0.2*(x/100)};
    \addlegendentry{\JE{提案手法}{Proposed}}
  \end{axis}
\fi
\end{tikzpicture}
\end{document}
